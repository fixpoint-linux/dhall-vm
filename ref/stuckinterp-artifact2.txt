static void print_text_escaped(FILE *out, const char *s) {
    for (const char *p = s; *p; p++) {
        switch (*p) {
        case '"': fputs("\\\"", out); break;
        case '\\': fputs("\\\\", out); break;
        case '\n': fputs("\\n", out); break;
        case '\t': fputs("\\t", out); break;
        case '\r': fputs("\\r", out); break;
        default: fputc(*p, out); break;
        }
    }
}
